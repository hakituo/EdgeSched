    \textbf{Generic Phase Profile --- No Model-Name Routing}\\[3pt]
    {\small \textbf{Service semantics}: \texttt{RESIDENT} $\mid$
    \texttt{BURST} $\mid$ \texttt{SESSION\_WARM} $\mid$ \texttt{HELPER}}\\[2pt]
    {\small \textbf{Placement capability}: CPU $\mid$ GPU $\mid$
    validated hybrid placement (layer/component/state)}\\[2pt]
    {\small \textbf{Execution controls}: queueability, preemption, batching,
    dependencies, and SM controllability}\\[2pt]
    {\scriptsize \textbf{Measured profiles}: DDR/VRAM, TTFT/TTFA/RTF,
    SM scaling, interference, and transition costs}
};

    effective SM allocation, bandwidth pressure, and recent transitions}

\node[decision] (capacity) at (-3.15,3.65) {
    \textbf{1. Capacity and Residency Planner}\\[3pt]
    {\small choose CPU/GPU/hybrid residency $r_i$}\\[2pt]
    {\scriptsize enforce DDR/VRAM safety; keep the resident service reachable;
    include load, migration, KV/state, and restoration costs}
};

\node[decision] (compute) at (3.15,3.65) {
    \textbf{2. Compute-Share Scheduler}\\[3pt]
    {\small choose CPU cores $c_i$, effective SMs $s_i$, and batch $b_i$}\\[2pt]
    {\scriptsize use only verified profiles; protect interaction/audio deadlines
    while bounding diffusion slowdown}
};

\node[action] (action) at (0,0.75) {
    \textbf{Event-Driven Joint Action}\\[2pt]
    {$a_t=\bigl(\{(r_i,c_i,s_i,b_i)\}_{i\in\mathcal A_t},\mu\bigr)$}\\[1pt]
    {\scriptsize $\mu$: hold $\mid$ urgent release $\mid$ atomic restore $\mid$
    reconfigure; switch only when predicted gain exceeds transition cost and hysteresis}
};
